    <script>
        // 1. 영화 데이터 (20개)
        const movies = [
            { id: 1, title: "파묘", color: "#2c3e50" },

            { id: 14, title: "기생충", color: "#636e72" },
            { id: 15, title: "인터스텔라", color: "#2980b9" },
            { id: 16, title: "인셉션", color: "#2d3436" },
            { id: 17, title: "아바타 2", color: "#00cec9" },
            { id: 18, title: "스파이더맨", color: "#d63031" },
            { id: 19, title: "가오갤 3", color: "#10ac84" },
            { id: 20, title: "엘리멘탈", color: "#fdcb6e" }
        ];

        const moviePool = document.getElementById('movie-pool');

        // 2. 영화 카드 생성
        movies.forEach(movie => {
            const card = document.createElement('div');
            card.className = 'movie-card';
            card.draggable = true;
            card.id = 'movie-' + movie.id;
            card.style.backgroundColor = movie.color;
            card.innerText = movie.title;
            
            card.addEventListener('dragstart', drag);
            moviePool.appendChild(card);
        });

        // 3. 드래그 앤 드롭 로직
        function allowDrop(ev) {
            ev.preventDefault();
        }

        function drag(ev) {
            ev.dataTransfer.setData("text", ev.target.id);
        }

        function drop(ev) {
            ev.preventDefault();
            const data = ev.dataTransfer.getData("text");
            const draggedElement = document.getElementById(data);
            
            let target = ev.target;
            // 드롭 위치가 카드가 아닌 영역(컨텐츠 박스)을 찾음
            if (!target.classList.contains('tier-content') && target.id !== 'movie-pool') {
                target = target.closest('.tier-content') || target.closest('#movie-pool');
            }
            
            if (target) {
                target.appendChild(draggedElement);
            }
        }

        // 4. 이미지 저장 기능
        function saveImage() {
            const container = document.getElementById('tier-container');
            const btn = document.querySelector('.save-btn');
            btn.innerText = "저장 중...";
            
            html2canvas(container).then(canvas => {
                const link = document.createElement('a');
                link.download = '나의_영화_티어리스트.png';
                link.href = canvas.toDataURL('image/png');
                link.click();
                btn.innerText = "이미지로 저장하기";
            }).catch(err => {
                console.error(err);
                alert("이미지 저장에 실패했습니다.");
                btn.innerText = "이미지로 저장하기";
            });
        }
    </script>
</body>
</html>

3. 영화를 드래그하여 배치하고 **'이미지로 저장하기'**를 눌러 결과를 확인하세요!
